# 설정 폴더에서 실행
cd 설정

# 1. 의존성 설치
npm install

# 2. 전체 토큰 빌드 (CSS/SCSS/Swift/Kotlin/device/radius)
npm run build

# 개별 플랫폼 빌드
npm run build:css
npm run build:scss
npm run build:ios
npm run build:android
